## 实现路径
bsim4_eval()中(~line 200):
  T = c->temp; Tnom = 300.15f; deltaT = T - Tnom; T_ratio = T/Tnom;
  vth0_T = vth0 + kt1*deltaT/Tnom + kt2*(deltaT/Tnom)^2
  u0_T = u0 * powf(T_ratio, ute)
  ua_T = ua + ua1*(T_ratio-1); ub_T = ub + ub1*(T_ratio-1); uc_T = uc + uc1*(T_ratio-1)
